    コードの読み方

    // ① 部品を作る
var picker = ColorPicker.create({
  mode: 'inline',                      // 'dialog' にすると小窓になる
  host: document.getElementById('場所'),
  color: '#e04f39',
  onChange: function (hex) {           // 人が操作したときだけ呼ばれる
    state.fill = hex;  paint();
  }
});

// ② 外から色を変える(onChange は呼ばれないので、画面の更新は、自分でする)
picker.setColor('#378add');
state.fill = '#378add';  paint();

// ③ 今の色を読む
var hex = picker.getColor();

// ④ 言語を切り替える
picker.setTexts({ hue: '色相', sat: '彩度', lum: '明るさ', palette: '色の見本',
                  hex: '色コード', rgb: 'RGB', dec: '減らす', inc: '増やす', close: '閉じる' });
picker.setTitle('図形の塗りの色');

// ⑤ ダイアログのとき
picker.open();  picker.close();  picker.isOpen();
